cmake_minimum_required(VERSION 3.22.1)
project(donuthle_android)

add_library(donuthle SHARED native_bridge.cpp)
find_library(log-lib log)
find_library(gles2-lib GLESv2)
set(rust_core "${CMAKE_CURRENT_LIST_DIR}/prebuilt/${ANDROID_ABI}/libdonuthle.a")
if(EXISTS "${rust_core}")
    add_library(donuthle_core STATIC IMPORTED)
    set_target_properties(donuthle_core PROPERTIES IMPORTED_LOCATION "${rust_core}")
    target_link_libraries(donuthle ${log-lib} ${gles2-lib} donuthle_core android log z dl m)
else()
    target_compile_definitions(donuthle PRIVATE DONUTHLE_NO_CORE=1)
    target_link_libraries(donuthle ${log-lib} ${gles2-lib})
endif()
